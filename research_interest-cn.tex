\begin{rubric}{研究兴趣}
{\large
\entry*[]{\textbf{复杂三维生物结构的医学与电镜图像分析：} 面向线粒体、细胞核及多细胞器等复杂三维生物结构，研究其在 CT、PET 与EM数据中的表示、分割与实例解析问题。}
\entry*[]{\textbf{弱/少监督条件下的三维实例分割方法：} 针对高标注成本与跨数据分布差异问题，研究一致性正则化、对比学习与结构先验相结合的弱/半监督学习方法，以提升三维医学图像与 EM 图像分割的鲁棒性与泛化能力。}
\entry*[]{\textbf{全细胞器分割基准与大规模数据集构建：} 牵头构建 MitoEM 2.0 等三维实例分割基准，探索“方法—数据集—评测”协同推进的研究范式，支撑可复现、可比较的算法评测与方法发展。}
\entry*[]{\textbf{面向电镜图像的结构感知模型与跨尺度学习：} 探索 skeleton/geometry 先验、多尺度表示与 mixture-of-experts 等机制，推进面向三维 EM 数据的实例分割模型与基础模型研究。}
}
\end{rubric}
